  bedrijf              String?

model Deskcontrole {
  id                       Int                        @id @default(autoincrement())
  attestId                 String                     @unique @map("attest_id") @db.Uuid
  auditeur                 String
  lidId                    Int                        @map("lid_id")
  lid                      Lid                        @relation(fields: [lidId], references: [id], onDelete: Restrict)
  procescertificaatId      Int                        @map("procescertificaat_id")
  procescertificaat        Procescertificaat          @relation(fields: [procescertificaatId], references: [id], onDelete: Restrict)
  linkAttest               String                     @unique @map("link_attest")
  attestnummer             String                     @unique
  status                   DeskcontroleStatus         @default(GEEN)
  deadlineSanctie          DateTime                   @map("deadline_sanctie") @db.Date
  mailSanctieVerzonden     Boolean                    @default(false) @map("mail_sanctie_verzonden")
  typeControle             DeskcontroleType           @map("type_controle")
  deadlineCorrectie        DateTime?                  @map("deadline_correctie") @db.Date
  mailCorrectieVerzonden   Boolean                    @default(false) @map("mail_correctie_verzonden")
  oneDrive                 String?                    @map("onedrive_url")
  voorwaardelijkeOpheffing Boolean                    @default(false) @map("voorwaardelijke_opheffing")
  opmerkingen              String?                    @db.Text
  datumControle            DateTime                   @map("datum_controle") @db.Date
  adres                    String?
  finalisatieDatum         DateTime?                  @map("finalisatie_datum") @db.Date
  aangemaaktOp             DateTime                   @default(now()) @map("aangemaakt_op")
  bijgewerktOp             DateTime                   @updatedAt @map("bijgewerkt_op")
  verwijderdOp             DateTime?                  @map("verwijderd_op")
  vaststellingen           DeskcontroleVaststelling[]

  @@index([lidId])
  @@index([procescertificaatId])
  @@index([status])
  @@index([typeControle])
  @@index([datumControle])
  @@index([deadlineSanctie])
  @@index([deadlineCorrectie])
  @@index([finalisatieDatum])
  @@index([verwijderdOp])
  @@map("deskcontroles")
}

  excelRij            Int          @map("excel_rij")

  ncId                String       @map("nc_id")
  omschrijving        String?      @db.Text
  vastgesteldDoorCi   String?      @map("vastgesteld_door_ci")

  motivatieAanpassing String?      @map("motivatie_aanpassing") @db.Text

  @@index([deskcontroleId])
  @@index([ncId])
  @@index([categorie])
  @@map("deskcontrole_vaststellingen")
}
